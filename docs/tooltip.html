Tooltip

CSS hover/focus tooltip, optional Esc JS. Depends on: motion.

PHP 1,153 B · Blade 262 B · JS 442 B
Add to library

Install

php vui add tooltip button

PHP

echo tooltip(raw(
    tooltip_trigger('Hover', 'tip', button_classes('outline')) .
    tooltip_content('Add to library', 'tip')
));

Blade

<x-ui.tooltip>
    <x-ui.tooltip.trigger for="tip" :class="button_classes('outline')">Hover</x-ui.tooltip.trigger>
    <x-ui.tooltip.content id="tip">Add to library</x-ui.tooltip.content>
</x-ui.tooltip>

components/tooltip/tooltip.php

<?php
function tooltip($c='',$class='',$attrs=[]){return el('span','tooltip',"group/tooltip relative inline-flex w-fit",$c,$class,$attrs);}
function tooltip_trigger($c='',$for='',$class='',$attrs=[]){return el('button','tooltip-trigger','',$c,$class,['type'=>'button','aria-describedby'=>$for?:null]+$attrs);}
function tooltip_content($c='',$id='',$class='',$attrs=[]){return el('span','tooltip-content',"invisible absolute start-1/2 bottom-full z-50 mb-2 w-max max-w-xs -translate-x-1/2 scale-0 rounded-md bg-foreground px-3 py-1.5 text-xs text-balance text-background opacity-0 transition-[visibility,scale,opacity] delay-[150ms,150ms,0s] duration-[0s,0s,150ms] group-hover/tooltip:visible group-hover/tooltip:scale-100 group-hover/tooltip:opacity-100 group-hover/tooltip:delay-0 group-has-[:focus-visible]/tooltip:visible group-has-[:focus-visible]/tooltip:scale-100 group-has-[:focus-visible]/tooltip:opacity-100 group-has-[:focus-visible]/tooltip:delay-0 rtl:translate-x-1/2",raw(h($c).'<span class="absolute inset-x-0 -bottom-1 mx-auto size-2.5 rotate-45 rounded-[2px] bg-foreground"></span>'),$class,['id'=>$id?:null,'role'=>'tooltip']+$attrs);}

js/tooltip.js

// vui tooltip (optional): Esc dismisses (WCAG 1.4.13). CSS alone can't do this.
addEventListener('keydown', e => { if (e.key == 'Escape') document.querySelectorAll('[role=tooltip]').forEach(t => { if (getComputedStyle(t).scale != '0') t.hidden = true }) })
for (const n of ['pointerout', 'focusout']) addEventListener(n, e => { const t = e.target.closest?.('[data-slot=tooltip]')?.querySelector('[role=tooltip]'); if (t) t.hidden = false })
