Context Menu

Right-click menu. context-menu.js opens it at the pointer. Depends on: dropdown-menu.

PHP 796 B · Blade 558 B · JS 970 B
Right click here

Install

php vui add context-menu

PHP

echo context_menu_trigger('Right click here', 'cm', 'flex h-36 w-72 items-center justify-center rounded-md border border-dashed text-sm');
echo context_menu_content(raw(
    context_menu_item(raw('Back' . context_menu_shortcut('⌘[')), 'cm') .
    context_menu_item('Reload', 'cm') .
    context_menu_separator() .
    context_menu_item('Delete', 'cm', 'destructive')
), 'cm', 'w-52');

Needs context-menu.js to open at the pointer. Without it the browser's own menu shows. Add dropdown-menu.js for arrow keys.

// Needs context-menu.js. Without it the browser's own menu shows.
function context_menu_trigger($c='',$for='',$class='',$attrs=[]){return el('div','context-menu-trigger','',$c,$class,['tabindex'=>0,'data-for'=>$for]+$attrs);}

function context_menu_item($c='',$for='',$variant='default',$class='',$attrs=[]){return dropdown_menu_item($c,$for,$variant,$class,$attrs);}
function context_menu_label($c='',$class='',$attrs=[]){return dropdown_menu_label($c,$class,$attrs);}
function context_menu_separator($class='',$attrs=[]){return dropdown_menu_separator($class,$attrs);}
function context_menu_shortcut($c='',$class='',$attrs=[]){return dropdown_menu_shortcut($c,$class,$attrs);}

js/context-menu.js

// vui context-menu: opens at the pointer, or the trigger for the keyboard menu key. Stays on screen, follows dir.
addEventListener('contextmenu', e => {
  const t = e.target.closest?.('[data-slot=context-menu-trigger]'), m = t && document.getElementById(t.dataset.for)
  if (!m) return
  e.preventDefault()
  const r = t.getBoundingClientRect()
  let x = e.clientX || r.left, y = e.clientY || r.top
  const open = () => {
    m.style.cssText = 'position:fixed;inset:auto;margin:0'
    if (!m.matches(':popover-open')) m.showPopover()
    if (getComputedStyle(m).direction == 'rtl') x -= m.offsetWidth
    m.style.left = Math.max(8, Math.min(x, innerWidth - m.offsetWidth - 8)) + 'px'
    m.style.top = Math.max(8, Math.min(y, innerHeight - m.offsetHeight - 8)) + 'px'
  }
  // Linux and macOS fire this on mouse down: the mouse up would light-dismiss a popover opened now.
  e.buttons ? addEventListener('pointerup', () => setTimeout(open), { once: true }) : open()
})
