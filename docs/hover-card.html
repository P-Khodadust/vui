Hover Card

CSS hover and focus preview card. Depends on: motion.

PHP 1,025 B · Blade 248 B · JS 0 B
@nextjsThe React framework, created and maintained by Vercel.

Install

php vui add hover-card button

PHP

echo hover_card(raw(
    hover_card_trigger('@nextjs', '#', button_classes('link')) .
    hover_card_content('The React framework, created and maintained by Vercel.')
));

Blade

<x-ui.hover-card>
    <x-ui.hover-card.trigger href="#" :class="button_classes('link')">@@nextjs</x-ui.hover-card.trigger>
    <x-ui.hover-card.content>The React framework, created and maintained by Vercel.</x-ui.hover-card.content>
</x-ui.hover-card>

components/hover-card/hover-card.php

<?php
function hover_card($c='',$class='',$attrs=[]){return el('span','hover-card',"group/hover-card relative inline-flex w-fit",$c,$class,$attrs);}
function hover_card_trigger($c='',$href='#',$class='',$attrs=[]){return el('a','hover-card-trigger','',$c,$class,['href'=>$href]+$attrs);}
// scale-0: a hidden card must not widen the page. delay-150: the pointer can reach the card, even diagonally.
function hover_card_content($c='',$class='',$attrs=[]){return el('span','hover-card-content',"invisible absolute start-0 top-full z-50 mt-1 w-64 scale-0 rounded-md border bg-popover p-4 text-popover-foreground opacity-0 shadow-md outline-hidden transition-[visibility,scale,opacity] delay-[150ms,150ms,0s] duration-[0s,0s,150ms] group-hover/hover-card:visible group-hover/hover-card:scale-100 group-hover/hover-card:opacity-100 group-hover/hover-card:delay-300 group-has-[:focus-visible]/hover-card:visible group-has-[:focus-visible]/hover-card:scale-100 group-has-[:focus-visible]/hover-card:opacity-100",$c,$class,$attrs);}
