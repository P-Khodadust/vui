Input Group

Input with icons, text or buttons inside. Depends on: button, input.

PHP 1,556 B · Blade 364 B · JS 0 B

Install

php vui add input-group icon

PHP

echo input_group(raw(
    input_group_input(attrs: ['placeholder' => 'Search', 'aria-label' => 'Search']) .
    input_group_addon(raw(icon('m21 21-4.34-4.34M19 11a8 8 0 1 1-16 0 8 8 0 0 1 16 0'))) .
    input_group_addon(raw(input_group_button('Go')), 'inline-end')
), 'max-w-sm');

<?php
function input_group($c='',$class='',$attrs=[]){return el('div','input-group',"relative flex h-9 w-full min-w-0 items-center rounded-md border border-input shadow-xs transition-[color,box-shadow] outline-none dark:bg-input/30 has-[>[data-align=inline-start]]:[&>input]:ps-2 has-[>[data-align=inline-end]]:[&>input]:pe-2 has-[input:focus-visible]:border-ring has-[input:focus-visible]:ring-[3px] has-[input:focus-visible]:ring-ring/50 has-[[aria-invalid=true]]:border-destructive has-[[aria-invalid=true]]:ring-destructive/20 dark:has-[[aria-invalid=true]]:ring-destructive/40",$c,$class,['role'=>'group']+$attrs);}
function input_group_addon($c='',$align='inline-start',$class='',$attrs=[]){return el('div','input-group-addon',cn("flex h-auto cursor-text items-center justify-center gap-2 py-1.5 text-sm font-medium text-muted-foreground select-none [&>svg:not([class*='size-'])]:size-4",$align=='inline-end'?"order-last pe-3 has-[>button]:me-[-0.45rem]":"order-first ps-3 has-[>button]:ms-[-0.45rem]"),$c,$class,['data-align'=>$align,'onclick'=>"event.target.closest('button')||this.parentNode.querySelector('input').focus()"]+$attrs);}
function input_group_button($c='',$variant='ghost',$class='',$attrs=[]){return button($c??'',$variant,'sm',cn("h-6 gap-1 rounded-[calc(var(--radius)-5px)] px-2 shadow-none! has-[>svg]:px-2",$class),['type'=>'button']+$attrs);}
function input_group_input($class='',$attrs=[]){return input(cn("flex-1 rounded-none border-0! bg-transparent shadow-none! focus-visible:ring-0! dark:bg-transparent!",$class),$attrs);}
