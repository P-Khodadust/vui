Input OTP

One-time code boxes on one native input.

PHP 921 B · Blade 128 B · JS 0 B

echo input_otp(6, attrs: ['name' => 'code', 'aria-label' => 'One-time code']);

<?php
// One native input drawn as boxes: paste, delete, autofill and the SMS code suggestion just work. $attrs go on the input.
function input_otp($length=6,$class='',$attrs=[]){return el('span','input-otp',"inline-block w-[calc(var(--n)*2.25rem+2px)] overflow-clip rounded-md border border-input shadow-xs transition-all has-focus-visible:border-ring has-focus-visible:ring-[3px] has-focus-visible:ring-ring/50 has-aria-invalid:border-destructive has-disabled:opacity-50 dark:bg-input/30",raw('<input'.attrs($attrs+['class'=>"block h-[calc(2.25rem-2px)] w-[calc(var(--n)*2.25rem+2.25rem)] bg-[repeating-linear-gradient(90deg,transparent_0_calc(2.25rem-1px),var(--input)_0_2.25rem)] ps-[calc(1.125rem-.5ch)] font-mono text-sm tracking-[calc(2.25rem-1ch)] outline-none",'inputmode'=>'numeric','autocomplete'=>'one-time-code','pattern'=>'[0-9]*','maxlength'=>$length]).'>'),$class,['dir'=>'ltr','style'=>"--n:$length"]);}
