Toast

Notifications. Server-rendered, toast.js adds window.toast() and auto-dismiss. Depends on: icon, motion.

PHP 1,045 B · Blade 234 B · JS 761 B

Install

php vui add toast button

PHP

echo button('Show toast', 'outline', attrs: ['onclick' => "toast('Event has been created')"]);
echo toaster();

<x-ui.button variant="outline" onclick="toast('Event has been created')">Show toast</x-ui.button>

components/toast/toast.php

<?php
function toast($c='',$variant='default',$close='Close',$class='',$attrs=[]){
return el('div','toast',"flex w-full items-center justify-between gap-2 rounded-md border bg-popover p-4 text-sm text-popover-foreground shadow-lg data-[variant=destructive]:text-destructive",raw('<span data-slot="toast-title">'.h($c).'</span>'.el('button','toast-close',"shrink-0 rounded-xs opacity-70 hover:opacity-100",raw(icon('M18 6 6 18M6 6l12 12')),'',['type'=>'button','aria-label'=>$close,'onclick'=>'this.parentNode.dataset.out=1;setTimeout(()=>this.parentNode.remove(),200)'])),$class,['data-variant'=>$variant,'role'=>$variant=='destructive'?'alert':null]+$attrs);
}
// The template lets toast.js add toasts without any class strings in JS.
function toaster($c='',$close='Close',$class='',$attrs=[]){
return el('section','toaster',"fixed end-4 bottom-4 z-50 flex w-[calc(100%-2rem)] max-w-[356px] flex-col gap-2",raw(h($c).'<template>'.toast('','default',$close).'</template>'),$class,$attrs+['aria-label'=>'Notifications','aria-live'=>'polite']);
}

js/toast.js

// vui toast: window.toast(message, variant, ms). Also auto-dismisses server-rendered toasts.
{
  const arm = (n, ms = 4000) => {
    let t
    const go = () => t = setTimeout(() => { n.dataset.out = 1; setTimeout(() => n.remove(), 200) }, ms)
    n.onpointerenter = () => clearTimeout(t)
    n.onpointerleave = go
    go()
  }
  window.toast = (m, v = 'default', ms) => {
    const box = document.querySelector('[data-slot=toaster]'), n = box.querySelector('template').content.firstChild.cloneNode(true)
    n.querySelector('[data-slot=toast-title]').textContent = m
    n.dataset.variant = v
    if (v == 'destructive') n.role = 'alert'
    box.append(n)
    arm(n, ms)
    return n
  }
  document.querySelectorAll('[data-slot=toast]').forEach(n => arm(n))
}
