Button Group

Join related buttons and inputs. Depends on: separator.

PHP 1,054 B · Blade 316 B · JS 0 B
https://

Install

php vui add button-group button input

PHP

echo button_group(raw(button('Archive', 'outline') . button('Report', 'outline') . button('Snooze', 'outline')));
echo button_group(raw(button('Copy', 'secondary') . button_group_separator() . button('Paste', 'secondary')));
echo button_group(raw(button_group_text('https://') . input('', ['aria-label' => 'Address'])));

<?php
function button_group($c='',$orientation='horizontal',$class='',$attrs=[]){return el('div','button-group',cn("flex w-fit items-stretch has-[>[data-slot=button-group]]:gap-2 [&>*]:focus-visible:relative [&>*]:focus-visible:z-10 [&>input]:flex-1",$orientation=='vertical'?"flex-col [&>*:not(:first-child)]:rounded-t-none [&>*:not(:first-child)]:border-t-0 [&>*:not(:last-child)]:rounded-b-none":"[&>*:not(:first-child)]:rounded-s-none [&>*:not(:first-child)]:border-s-0 [&>*:not(:last-child)]:rounded-e-none"),$c,$class,['role'=>'group','data-orientation'=>$orientation]+$attrs);}
function button_group_text($c='',$class='',$attrs=[]){return el('div','button-group-text',"flex items-center gap-2 rounded-md border bg-muted px-4 text-sm font-medium shadow-xs [&_svg]:pointer-events-none [&_svg:not([class*='size-'])]:size-4",$c,$class,$attrs);}
function button_group_separator($orientation='vertical',$class='',$attrs=[]){return separator($orientation,true,cn("relative m-0! self-stretch bg-input data-[orientation=vertical]:h-auto",$class),$attrs);}
